```bash
ls -l
ls -lt        # 最近 → 最远
ls -ltr       # 最远 → 最近（r = reverse）
ls -lt --full-time   # 显示完整时间戳
ls -lt | head -n 10   # 最近 10 个文件

ls -a #显示隐藏文件

ls -alt # 显示所有文件（包括隐藏文件），并按照修改时间从最远到最近排序
```
